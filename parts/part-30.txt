IVE = cached.rows;
        return cached.rows;
      }
    } catch(e){}
    const parts = await Promise.all(WEST_FEEDS.map(function(pair){
      return fetch(pair[1]).then(function(r){ if (!r.ok) throw new Error(pair[0]); return r.text(); }).then(function(text){ return parseWestM3U(text, pair[0]); }).catch(function(){ return []; });
    }));
    const rows = bestWest([].concat.apply([], parts));
    if (rows.length){
      window.__LIVE = rows;
      try { localStorage.setItem("goar-west-v2", JSON.stringify({ at: Date.now(), rows: rows })); } catch(e){}
    }
    return window.__LIVE || [];
  })();
  return westPromise;
}
function liveCatalog(){
  if (window.__LIVE) return window.__LIVE;
  const el = document.getElementById("goar-live");
  window.__LIVE = el ? JSON.parse(el.textContent) : [];
  return window.__LIVE;
}
function liveCountries(rows){
  const counts = {};
  rows.forEach(function(ch){ counts[ch.c] = (counts[ch.c] || 0) + 1; });
  return Object.keys(counts).filter(function(c){ return counts[c] >= 8; }).sort(function(a,b){ return counts[b]-counts[a]; });
}
function liveArtFallback(img){
  if (!img || !img.parentNode || img.dataset.gone) return;
  img.dataset.gone = "1";
  const card = img.closest(".live-card");
  const raw = card && card.querySelector("b") ? card.querySelector("b").textContent : "";
  const mark = raw.replace(/\s*\([^)]*\)/g, "").trim().slice(0, 2).toUpperCase() || "TV";
  const ph = document.createElement("div");
  ph.className = "live-ph";
  ph.textContent = mark;
  img.replaceWith(ph);
}
function paintLive(){
  const all = liveCatalog();
  const q = liveQuery.trim().toLowerCase();
  const rows = all.filter(function(ch){
    if (liveCountry !== "All" && ch.c !== liveCountry) return false;
    if (!q) return true;
    return (ch.n + " " + ch.c + " " + (ch.g || "")).toLowerCase().indexOf(q) >= 0;
  }).sort(function(a,b){ return (b.r||0)-(a.r||0) || a.n.localeCompare(b.n); });
  const box = document.getElementById("liveGrid");
  const meta = document.getElementById("liveMeta");
  if (meta) meta.textContent = rows.length + " with artwork" + (liveCountry === "All" ? "" : " · " + liveCountry);
  const slice = rows.slice(0, liveLimit);
  if (!slice.length){ box.innerHTML = '<p class="live-empty">Nothing in this filter.</p>'; return; }
  box.innerHTML = slice.map(function(ch, i){
    const res = ch.r ? (ch.r + "p") : "Live";
    const name = String(ch.n).replace(/&/g,"&").replace(/</g,"<");
    const sub = String(ch.c + " · " + (ch.g || "Live")).replace(/&/g,"&").replace(/</g,"<");
    return '<button type="button" class="live-card" data-i="' + i + '"><img src="' + ch.i + '" alt="" loading="lazy" referrerpolicy="no-referrer"><b>' + name + '</b><span>' + sub + '</span><em>' + res + '</em></button>';
  }).join("");
  box.querySelectorAll(".live-card").forEach(function(btn){
    const img = btn.querySelector("img");
    img.onerror = function(){ liveArtFallback(img); };
    img.onload = function(){
      if (!img.naturalWidth || (img.naturalWidth === 161 && img.naturalHeight === 81)) liveArtFallback(img);
    };
    if (img.complete) img.onload();
    btn.onclick = function(){ playWest(slice[Number(btn.getAttribute("data-i"))]); };
  });
  let more = document.getElementById("liveMore");
  if (rows.length > slice.length){
    if (!more){
      more = document.createElement("button");
      more.id = "liveMore";
      more.type = "button";
      more.className = "live-more";
      box.parentNode.appendChild(more);
    }
    more.hidden = false;
    more.textContent = "Show more · " + (rows.length - slice.length) + " left";
    more.onclick = function(){ liveLimit += 80; paintLive(); };
  } else if (more) more.hidden = true;
}

function setActiveTab(tab){
  activeTab = tab;
  document.querySelectorAll("nav a[data-tab]").forEach(x => x.classList.toggle("active", x.dataset.tab === tab));

}
function routeTo(tab){
  const gridOpen = document.getElementById("gridView").classList.contains("open");
  if (activeTab === tab && !document.body.classList.contains("prov-mode") && !gridOpen) return;
  setActiveTab(tab);
  const main = document.getElementById("mainContent");
  main.style.transition = "opacity .15s";
  main.style.opacity = "0.3";
  setTimeout(() => {
    if (tab==="movie") buildMovieTab();
    else if (tab==="tv") buildTVTab();
    else if (tab==="anime") buildAnimeTab();
    else if (tab==="kids") buildKidsTab();
    else if (tab==="music") buildMusicTab();
    else if (tab==="live") buildWestLive();
    else if (tab==="hubs") buildHubsTab();
    else if (tab==="list") buildListTab();
    else buildHome();
    main.style.opacity = "1";
    window.scrollTo({ top: 0, behavior: "smooth" });
  }, 60);
}
